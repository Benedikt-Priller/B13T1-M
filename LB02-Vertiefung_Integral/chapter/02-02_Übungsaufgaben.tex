\begin{Exercise}{S. 54/10}{}
\begin{enumerate}[leftmargin=*]
    \item [10.] {
        \begin{enumerate}[label=\alph*),leftmargin=*]
            \item [a)] {
                $\displaystyle \int_{-4}^{3} f(x) \ \mathrm{d}x = -\SI{2.86}{}$\\[2.0ex]
                Es gilt: \quad \begin{minipage}[t]{0.85\linewidth}
                    \begin{itemize}[leftmargin=*]
                        \item $A_{1}$ ist die Fläche, die $G_{f}$ mit der x-Achse im Intervall $\left[ \ -4; 0 \ \right]$ einschließt
                        \item $A_{2}$ ist die Fläche, die $G_{f}$ mit der x-Achse im Intervall $\left[ \ 0; 3 \ \right]$ einschließt
                    \end{itemize}
                \end{minipage}\\[2.0ex]
                Da $A_{1} > A_{2}$ und $A_{1}$ unter der x-Achse liegt, folgt eine negative Flächenbilanz\\
                }
            \item [b)] {
                $\displaystyle \int_{0}^{3} f(x) \ \mathrm{d}x = \SI{2.475}{}$\\
                \begin{align*}
                    \int_{-4}^{0} f(x) \ \mathrm{d}x &= \int_{-4}^{3} f(x) \ \mathrm{d}x - \int_{0}^{3} f(x) \ \mathrm{d}x \\[1.0ex]
                    &= -\SI{2.86}{} - \SI{2.475}{} \\[1.0ex]
                    &\uuline{= -\SI{5.335}{}}
                \end{align*}
                }
            \item [c)] {
                Lage der Nullstellen von $\displaystyle \int_{2}^{x} f(t) \ \mathrm{d}t = F_{2}(x)$:
                \begin{itemize}
                    \item $x_{1} < -4$
                    \item $-4 < x_{2} < 0$
                    \item $x_{3} = 0$
                    \item $x_{4} > 3$
                \end{itemize}
                }
            \item [d)] {
                \begin{align*}
                    g(x) &= -\dfrac{4}{3}x \cdot \left(x-3\right) &&\\[1.5ex]
                    &= -\dfrac{4}{3}\cdot \left(x^{2} -3x\right) &&
                \end{align*}\\[-2.0ex]
                \begin{align*}
                    \int_{0}^{3} g(x) \ \mathrm{d}x &= \left[ -\dfrac{4}{3}\cdot \left(\dfrac{1}{3}x^{3} - \dfrac{3}{2}x^{2}\right)\right]_{0}^{3} &&\\[1.5ex]
                    &= -\dfrac{4}{3} \cdot \left[\dfrac{1}{3} \cdot 3^{3} - \dfrac{3}{2} \cdot 3^{2} - \left(\dfrac{1}{3} \cdot 0^{3} - \dfrac{3}{2} \cdot 0^{2}\right)\right]&&\\[1.5ex]
                    &\uuline{= \SI{6}{}} &&
                \end{align*}
                }
            \item [e)] {
                \begin{align*}
                    A &=  \int_{0}^{3} g(x) \ \mathrm{d}x - \int_{0}^{3} f(x) \ \mathrm{d}x &&\\[1.5ex]
                    &= \SI{6}{} - \SI{2.475}{} &&\\[1.5ex]
                    &\uuline{= \SI{3.525}{}} &&
                \end{align*}
            }
        \end{enumerate}
        }
\end{enumerate}
\end{Exercise}
