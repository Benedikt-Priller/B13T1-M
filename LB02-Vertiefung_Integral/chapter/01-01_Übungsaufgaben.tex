                        f'(x) &= -\dfrac{1}{2} x - 1 &&

                    \end{align*}\\[-2.0ex]
                    \begin{tabularx}{0.95\linewidth}{c|X|X|X}

                        $G_{f}$ & $\nearrow$ & HoP & $\searrow$ \\           
                    \end{tabularx}\\[3.0ex]
