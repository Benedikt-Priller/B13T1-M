\begin{Exercise}{S. 20/1}{}
    \begin{enumerate}
        \item [1.]\begin{enumerate}[label=\alph*)]
            \item[a)] {
                $f(x) = x^{\frac{5}{4}}; \qquad D_{f} = \left] \ 0; \infty \ \right[$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{5}{4} \cdot x^{\frac{1}{4}} &&\\[-2.0ex]
                \end{align*}
                }
            \item [b)] {
                $f(x) = \ln(x + 2) + 3; \qquad D_{f} = \left] \ -2; \infty \ \right[$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{\left(x + 2\right) \cdot 1} = \dfrac{1}{x + 2} &&\\[-2.0ex]
                \end{align*}
                }
            \item [c)] {
                $f(x) = 3 \cdot ln(5x); \qquad D_{f} = \left] \ 0; \infty \ \right[$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= 3 \cdot \dfrac{1}{5x} \cdot 5 = \dfrac{3}{x} &&\\[-2.0ex]
                \end{align*}
                }
            \item [d)] {
                $f(x) = \sqrt[6]{x}; \qquad D_{f} = \left] \ 0; \infty \ \right[$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{6} \cdot x^{-\frac{5}{6}}&&\\[-2.0ex]
                \end{align*}
                }
            \item [e)] {
                $f(x) = 5x \cdot \ln(-x); \qquad D_{f} = \left] \ -\infty; -\dfrac{1}{2} \ \right[$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= 5 \cdot \ln(-x) + 5x \cdot \dfrac{1}{-x} \cdot (-1) = 5 \cdot \left[\ln(-x) + 1\right] &&\\[-2.0ex]
                \end{align*}
                }
            \item [f)] {
                $f(x) = e^{3x} - 4; \qquad D_{f} = \left] \ -\infty;  0 \ \right]$\\[-2.0ex]
                \begin{align*}
                    f'(x) &= e^{3x} \cdot 3 = 3e^{3x} &&\\[-2.0ex]
                \end{align*}
                }
        \end{enumerate}
    \end{enumerate}
\end{Exercise}

\begin{Exercise}{S. 20/2}{}
    \begin{enumerate}
        \item [2.]\begin{enumerate}[label=\alph*)]
            \item[a)] {
                $f(x) = \dfrac{e^{2x}+ 4}{e^{x}}; \qquad P(0/5) \in G_{f}; \qquad Q(5/0) \in G_{f^{-1}}$\\[-2.0ex]
                \begin{itemize}[leftmargin=*]
                    \item Tangentengleichung: \\[2.0ex]{
                        $f_{T}(x): \quad y = m \cdot x + t = \left(f^{-1}(x)\right)' \cdot x + t$\\
                        \vspace{-2.0ex}
                        \begin{align*}
                            \left(f^{-1}(x)\right)' &= \dfrac{1}{f'\left(f^{-1}(x)\right)} &&
                        \end{align*}\\[-2.0ex]
                        \begin{align*}
                            f'(x) &= \dfrac{e^{x} \cdot e^{2x} \cdot 2 - \left(e^{2x} + 4 \right) \cdot e^{x} \cdot 1}{\left(e^{x}\right)^2} &&\\[1.5ex]
                            &= \dfrac{e^{x} \cdot \left[2 \cdot e^{2x} - \left(e^{2x} + 4 \right) \right]}{\left(e^{x}\right)^2} &&\\[1.5ex]
                            &= \dfrac{e^{2x} - 4}{e^{x}} &&
                        \end{align*}\\[-2.0ex]
                        $Q$ in $f_{T}$: \\[-2.0ex]
                        \begin{align*}
                            0 &= \dfrac{1}{f'\left(f^{-1}(5)\right)} \cdot 5 + t &&\\[1.5ex]
                            0 &= \dfrac{1}{f'(0)} \cdot 5 + t &&\\[1.5ex]
                            0 &= 1 \cdot \dfrac{e^{0}}{e^{2 \cdot 0} - 4} \cdot 5 + t &&\\[1.5ex]
                            0 &= - \dfrac{1}{3} \cdot 5 + t &&\\[1.5ex]
                            0 &= -\dfrac{5}{3} + t &&| \quad + \dfrac{5}{3}\\[1.5ex]
                            t &= \dfrac{5}{3} &&
                        \end{align*}\\[-2.0ex]
                        $\Rightarrow \quad \uuline{f_{T}(x) = - \dfrac{1}{3} \cdot x + \dfrac{5}{3}}$\\
                        }
                \end{itemize}
                }
            \item[b)] {
                $f(x) = \ln(x^{2} - 3); \qquad P(2/0) \in G_{f}; \qquad Q(0/2) \in G_{f^{-1}}$\\[-2.0ex]
                \begin{itemize}[leftmargin=*]
                    \item Tangentengleichung: \\[2.0ex]{
                        $f_{T}(x): \quad y = \left(f^{-1}(x)\right)' \cdot x + t$\\
                        \vspace{-2.0ex}
                        \begin{align*}
                            \left(f^{-1}(x)\right)' &= \dfrac{1}{f'\left(f^{-1}(x)\right)} &&
                        \end{align*}\\[-2.0ex]
                        \begin{align*}
                            f'(x) &= \dfrac{1}{x^{2} - 3} \cdot 2x = \dfrac{2x}{x^{2} - 3} &&
                        \end{align*}\\[-2.0ex]
                        $Q$ in $f_{T}$: \\[-2.0ex]
                        \begin{align*}
                            2 &= \dfrac{1}{f'\left(f^{-1}(0)\right)} \cdot 0 + t &&\\[1.5ex]
                            2 &= \dfrac{1}{f'(2)} \cdot 0 + t &&\\[1.5ex]
                            2 &= 1 \cdot \dfrac{2^{2} - 3}{2 \cdot 2} \cdot 0 + t &&\\[1.5ex]
                            2 &= \dfrac{1}{4} \cdot 0 + t &&\\[1.5ex]
                            t &= 2 &&
                        \end{align*}\\[-2.0ex]
                        $\Rightarrow \quad \uuline{f_{T}(x) = \dfrac{1}{4} \cdot x + 2}$\\
                        }
                \end{itemize}
                }
            \item[c)] {
                $f(x) = \dfrac{x^2 + 2x}{3x -1}; \qquad P(1/f(1)) \in G_{f}; \qquad Q(f(1)/1) \in G_{f^{-1}}$\\[-2.0ex]
                \begin{itemize}[leftmargin=*]
                    \item Tangentengleichung: \\[2.0ex]{
                        $f_{T}(x): \quad y = \left(f^{-1}(x)\right)' \cdot x + t$\\
                        \vspace{-2.0ex}
                        \begin{align*}
                            \left(f^{-1}(x)\right)' &= \dfrac{1}{f'\left(f^{-1}(x)\right)} &&
                        \end{align*}\\[-2.0ex]
                        \begin{align*}
                            f'(x) &= \dfrac{\left(3x - 1\right) \cdot \left(2x + 2\right) - \left(x^2 + 2x\right) \cdot 3}{\left(3x - 1\right)^{2}} && \\[1.5ex]
                            &= \dfrac{3x^2 - 2x -2}{\left(3x - 1\right)^{2}} &&
                        \end{align*}\\[-2.0ex]
                        $Q$ in $f_{T}$: \\[-2.0ex]
                        \begin{align*}
                            1 &= \dfrac{1}{f'\left(f^{-1}(f(1))\right)} \cdot f(1) + t &&\\[1.5ex]
                            1 &= \dfrac{1}{f'(1)} \cdot f(1) + t &&\\[1.5ex]
                            1 &= 1 \cdot \dfrac{\left(3 \cdot 1 - 1\right)^{2}}{3 \cdot {1}^2 - 2 \cdot 1 -2} \cdot \dfrac{1^2 + 2 \cdot 1}{3 \cdot 1 -1} + t &&\\[1.5ex]
                            1 &= -4 \cdot \dfrac{3}{2} + t &&| \quad + 6\\[1.5ex]
                            t &= 7 &&
                        \end{align*}\\[-2.0ex]
                        $\Rightarrow \quad \uuline{f_{T}(x) = -4x + 7}$\\
                        }
                \end{itemize}
                }
            \item[d)] {
                $f(x) = e^{2x-4}-1; \qquad P(f^{-1}(0)/0) \in G_{f}; \qquad Q(0/f^{-1}(0)) \in G_{f^{-1}}$\\[-2.0ex]
                \begin{itemize}[leftmargin=*]
                    \item Umkehrfunktion: \\{
                        \vspace{-3.0ex}
                        \begin{align*}
                            x &= e^{2y -4} - 1 &&| \quad + 1\\[1.5ex]
                            x + 1 &= e^{2y -4} &&\\[1.5ex]
                            2y -4 &= \ln(x + 1) &&| \quad + 4\\[1.5ex]
                            2y &= \ln(x + 1) + 4 && | :2\\[1.5ex]
                            y &= \dfrac{1}{2} \cdot \ln(x + 1) + 2 &&
                        \end{align*}\\[-1.0ex]
                        $\Rightarrow \quad f^{-1}(x) = \dfrac{1}{2} \cdot \ln(x + 1) + 2$\\[-2.0ex]
                        }
                    \item Ableitung: \\[2.0ex]{
                        $f'(x) = e^{2x-4} \cdot 2$\\[-2.0ex]
                        }
                    \item Tangentengleichung: \\[2.0ex]{
                        $f_{T}(x): \quad y = \left(f^{-1}(x)\right)' \cdot x + t; \qquad \left(f^{-1}(x)\right)' = \dfrac{1}{f'\left(f^{-1}(x)\right)}$\\[2.0ex]
                        $Q$ in $f_{T}$: \\[-2.0ex]
                        \begin{align*}
                            f^{-1}(0) &= \dfrac{1}{f'\left(f^{-1}(0)\right)} \cdot 0 + t &&\\[1.5ex]
                            \dfrac{1}{2} \cdot \ln(0 + 1) + 2 &= \dfrac{1}{e^{2 \cdot \left(\dfrac{1}{2} \cdot \ln\left(0 + 1\right) + 2\right) -4} \cdot 2} \cdot 0 + t &&\\[2.0ex]
                            2 &= \dfrac{1}{2} \cdot 0 + t &&\\[1.5ex]
                            t &= 2 &&
                        \end{align*}\\[-2.0ex]
                        $\Rightarrow \quad \uuline{f_{T}(x) = \dfrac{1}{2} x + 2}$\\[-2.0ex]
                        }
                \end{itemize}
                }
        \end{enumerate}
    \end{enumerate}
\end{Exercise}
