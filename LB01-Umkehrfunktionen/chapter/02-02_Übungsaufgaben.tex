\begin{Exercise}{S. 32/4}{}
    \begin{enumerate}
        \item [4.]\begin{enumerate}[label=\alph*)]
            \item[a)] {
                $f(x) = \arctan\left(x^{2} + 7\right)$; \qquad $D_{f} = \mathbb{R}$ \\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{1 + \left(x^{2} + 7\right)^{2}} \cdot 2x &&\\[1.5ex]
                    &\uuline{= \dfrac{2x}{1 + \left(x^{2} + 7\right)^{2}}} && \\[1.5ex]
                \end{align*}
                }
            \item[b)] {
                $f(x) = \arctan\left(\sqrt[2]{x + \dfrac{1}{2}}\right)$; \qquad $D_{f} = \left[ \ -\dfrac{1}{2}; \infty \ \right[$ \\[-1.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{1 + \left(\sqrt[2]{x + \dfrac{1}{2}}\right)^{2}} \cdot \dfrac{1}{2} \cdot \left(x + \dfrac{1}{2}\right)^{-\frac{1}{2}} \cdot 1 &&\\[1.5ex]
                    &= \dfrac{1}{x + \dfrac{3}{2}} \cdot \dfrac{1}{2 \cdot \sqrt[2]{x + \dfrac{1}{2}}} && \\[1.5ex]
                    &= \dfrac{1}{\left(x + \dfrac{3}{2}\right) \cdot 2 \cdot \sqrt[2]{x + \dfrac{1}{2}}} && \\[1.5ex]
                    &\uuline{= \dfrac{1}{{\left(2x + 3\right) \cdot \sqrt[2]{x + \dfrac{1}{2}}}}} && 
                \end{align*}
                }
            \item[c)] {
                $f(x) = \arctan\left(e^{x}\right)$; \qquad $D_{f} = \mathbb{R}$ \\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{1 + \left(e^{x}\right)^{2}} \cdot e^{x} \cdot 1 &&\\[1.5ex]
                    &\uuline{= \dfrac{e^{x}}{e^{2x} + 1}} &&
                \end{align*}
                }
            \item[d)] {
                $f(x) = \arctan\left(x\right) + \arctan\left(\dfrac{1}{x}\right)$; \qquad $D_{f} = \mathbb{R} \setminus \{ 0 \}$ \\[-2.0ex]
                \begin{align*}
                    f'(x) &= \dfrac{1}{1 + \left(x\right)^{2}} \cdot 1 + \dfrac{1}{1 + \left(\dfrac{1}{x}\right)^{2}} \cdot \left(-1\right) \cdot \dfrac{1}{x^{2}} &&\\[1.5ex]
                    &= \dfrac{1}{x^{2} + 1} + \left(-1\right) \cdot  \dfrac{1}{1 + \dfrac{1}{x^{2}}} \cdot \dfrac{1}{x^{2}} && \\[1.5ex]
                    &= \dfrac{1}{x^{2} + 1} - \dfrac{1}{x^{2} + 1} && \\[1.5ex]
                    &\uuline{= 0} &&
                \end{align*}
                }
        \end{enumerate}
    \end{enumerate}
\end{Exercise}
