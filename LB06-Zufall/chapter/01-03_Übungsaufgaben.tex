\begin{Exercise}{S. 154/14}{}
\begin{enumerate}[leftmargin=*]
    \item [14.]{\begin{itemize}[leftmargin=*]
        \item {Frau: \qquad $E_{1} = \overline{\overline{M} \cup Z} = M \cap \overline{Z}$ \qquad $\Rightarrow$ \quad mit Milch ohne Zucker}\\[-1.5ex]
        \item {Tochter: \qquad $E_{2} = \overline{\overline{M} \cup \overline{Z}} = M \cap Z$ \qquad $\Rightarrow$ \quad mit Milch und Zucker}\\[-1.5ex]
        \item {Mann: \qquad $E_{3} = \overline{M \cap \overline{Z}} = \overline{M} \cup Z$ \qquad $\Rightarrow$ \quad ohne Milch oder mit Zucker}\\[-1.5ex]
        \end{itemize}
        Nur die Bestellung von Frau und Tochter sind eindeutig. Bei Herr Mayer gibt es mehrere Möglichkeiten, die zufriedenstellend sind.
        }
\end{enumerate}
\end{Exercise}

\begin{Exercise}{S. 156/ Test B - 2}{}
\begin{enumerate}[leftmargin=*]
    \item [2.]\begin{enumerate}[label=\alph*),leftmargin=*]
        \item[a)] {
            Nie wahr, da $A$ und $\overline{A}$ sich gegenseitig ausschließen und unvereinbar sind $A \cap \overline{A} = \emptyset$
            }
        \item[b)] {
            Immer wahr, da $A \in \Omega$ und die Vereinigungsmenge folglich immer $\Omega$ ist
            }
        \item[c)] {
            Nur wahr, wenn $A = \emptyset$ 
            }
        \item[d)] {
            Nur wahr, wenn $A = B$ (Ansonsten gilt: $\overline{A} \cup \overline{B} = \overline{A \cap B} \neq \overline{A \cup B}$)           
            }
    \end{enumerate}
\end{enumerate}
\end{Exercise}
