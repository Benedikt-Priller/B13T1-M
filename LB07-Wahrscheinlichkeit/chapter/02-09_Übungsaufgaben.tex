                $\overline{A}$ & \cellcolor{mylightgray} $\dfrac{50}{1000}$ & $\dfrac{250}{1000}$ & $\dfrac{300}{1000}$ \\ \hline
                $\Sigma$ &$\dfrac{570}{1000}$ & $\dfrac{430}{1000}$ & \cellcolor{mylightorange} $\dfrac{1000}{1000}$
            \end{tabularx}
            \vspace{0.5cm}
        \end{minipage}
        \hfill
        \begin{minipage}[t]{0.52\linewidth}
            $P(M \cap A) = \dfrac{520}{1000} = \SI{52}{\percent}$\\[1.5ex]
            Satz von Sylvester:
            \begin{align*}
                P(M \cup A) &= P(M) + P(A) - P(M \cap A) &&
            \end{align*}
            \begin{align*}
                \dfrac{520}{1000} + \dfrac{180}{1000} + \dfrac{50}{1000} &= \dfrac{570}{1000} + \dfrac{700}{1000} - \dfrac{520}{1000} &&\\[1.5ex]
                \dfrac{750}{1000} &= \dfrac{750}{1000} \quad \textit{(w)}&&
            \end{align*}
        \end{minipage}
        }
\end{enumerate}
\end{Exercise}
